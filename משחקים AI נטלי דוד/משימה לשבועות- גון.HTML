שָׁלוֹם גּוֹן! מְשִׂימָה אִינְטֶרְאַקְטִיבִית קוֹל נָשִׁי אֱנוֹשִׁי

מְגִילַּת רוּת: מַסָּע שֶׁל חֶסֶד וְתִקְוָה

קִרְאִי אֶת הַסִּפּוּר, הַקְשִׁיבִי לַשְּׁאֵלוֹת וְגִרְרִי אֶת הַתְּשׁוּבוֹת הַנְּכוֹנוֹת לִמְקוֹמָן!

מְהִירוּת:
מַקְרִיאָה בְּקוֹל נָשִׁי אֱנוֹשִׁי...

חֵלֶק א': הַסִּיפּוּר שֶׁלָּנוּ

שָׁנִים רַבּוֹת לִפְנֵי שֶׁנּוֹלַד דָּוִד הַמֶּלֶךְ, הָיָה רָעָב כָּבֵד בְּאֶרֶץ יִשְׂרָאֵל. לָאֲנָשִׁים לֹא הָיָה מָה לֶאֱכֹל.

אִישׁ עָשִׁיר אֶחָד מִבֵּית לֶחֶם, שֶׁקָּרְאוּ לוֹ אֱלִימֶלֶךְ, לֹא רָצָה לַעֲזוֹר לָעֲנִיִּים וְנָעַל אֶת הַשְּׁעָרִים שֶׁלּוֹ. הוּא לָקַח אֶת אִשְׁתּוֹ נָעֳמִי וְאֶת שְׁנֵי בָּנָיו, וּבָרַח לְאֶרֶץ מוֹאָב הַשְּׁכֵנָה.

בְּאֶרֶץ מוֹאָב הַמִּשְׁפָּחָה אִבְּדָה אֶת כָּל הַכֶּסֶף שֶׁלָּהּ, וְגַם אֱלִימֶלֶךְ וְהַבָּנִים מֵתוּ. נָעֳמִי נִשְׁאֲרָה לְבַדָּהּ בְּאֶרֶץ זָרָה וְהָיְתָה עֲצוּבָה מְאֹד.

הַכַּלָּה שֶׁלָּהּ, רוּת, אָהֲבָה אוֹתָהּ מְאֹד וְסֵירְבָה לַעֲזוֹב אוֹתָהּ. רוּת אָמְרָה לְנָעֳמִי מִילִּים יָפוֹת שֶׁל נֶאֱמָנוּת: "עַמֵּךְ עַמִּי וֵאלֹהַיִךְ אֱלֹהָי" – כְּלוֹמַר, אֲנִי הוֹלֶכֶת אִתָּךְ לְכָל מָקוֹם וְלֹא אֶעֱזוֹב אוֹתָךְ בַּצָּרָה.

הַשְּׁתַּיִם חָזְרוּ יַחַד לָעִיר בֵּית לֶחֶם בִּתְחִלַּת קְצִיר הַחִטִּים. בִּגְלַל שֶׁלֹּא הָיָה לָהֶן אֹכֶל בַּבַּיִת, רוּת הָלְכָה לַשָּׂדֶה הַחַקְלָאִי כְּדֵי לְלַקֵּט (לֶאֱסוֹף מֵהָרִצְפָּה) שִׁבֳּלִים בּוֹדְדוֹת שֶׁנָּפְלוּ לַקּוֹצְרִים. רוּת עָשְׂתָה חֶסֶד גָּדוֹל מְאֹד עִם חֲמוֹתָהּ נָעֳמִי.

הַהִתְקַדְּמוּת שֶׁל גּוֹן:
0 מִתּוֹךְ 5 שְׁאֵלוֹת פְּתוּרוֹת

חֵלֶק ב': מְשִׁימַת הַתְאָמָה

גִּרְרִי אֶת הַתְּשׁוּבָה הַמַּתְאִימָה מִסַּלְסִלַּת הַתְּשׁוּבוֹת מִשְּׂמֹאל (אוֹ לַחֲצִי עָלֶיהָ וְאָז עַל מִשְׁבֶּצֶת הַשְּׁאֵלָה)!

1

מַדּוּעַ עָזַב אֱלִימֶלֶךְ אֶת אֶרֶץ יִשְׂרָאֵל וְעָבַר לְאֶרֶץ מוֹאָב?

לַחֲצִי כָּאן אוֹ גִּרְרִי לְכָאן אֶת הַתְּשׁוּבָה

מָה קָרָה לְאֱלִימֶלֶךְ וְלִשְׁנֵי בָּנָיו בְּאֶרֶץ מוֹאָב?

לַחֲצִי כָּאן אוֹ גִּרְרִי לְכָאן אֶת הַתְּשׁוּבָה
3

הַעְתִּיקוּ מֵהַטֶּקְסְט אֶת הַמִּשְׁפָּט הַמְּפוּרְסָם שֶׁאָמְרָה רוּת לְנָעֳמִי כְּדֵי לְהַרְאוֹת לָהּ שֶׁהִיא נֶאֱמָנָה לָהּ.

לַחֲצִי כָּאן אוֹ גִּרְרִי לְכָאן אֶת הַתְּשׁוּבָה
4

מָה עָשְׂתָה רוּת בַּשָּׂדֶה כְּדֵי שֶׁיִּהְיֶה לָהּ וּלְנָעֳמִי מָה לֶאֱכֹל?

לַחֲצִי כָּאן אוֹ גִּרְרִי לְכָאן אֶת הַתְּשׁוּבָה

אֵילוּ תְּכוּנוֹת אֹפִי יֵשׁ לָרוּת עַל פִּי הַסִּפּוּר? (כִּתְבוּ 2 מִשְׁפָּטִים)

לַחֲצִי כָּאן אוֹ גִּרְרִי לְכָאן אֶת הַתְּשׁוּבָה

סַלְסִלַּת הַתְּשׁוּבוֹת

לַחֲצִי אוֹ גִּרְרִי

גּוֹן, קִרְאִי אֶת הַכַּרְטִיסִיּוֹת (אוֹ לַחֲצִי עַל הָרַמְקוֹל לִשְׁמֹעַ אוֹתָן), וְאָז הַתְאִימִי אוֹתָן לַשְּׁאֵלוֹת מִיָּמִין!

בִּגְלַל שֶׁהָיָה רָעָב כָּבֵד בְּאֶרֶץ יִשְׂרָאֵל, וְהוּא (שֶׁהָיָה עָשִׁיר) לֹא רָצָה לַעֲזוֹר לָעֲנִיִּים וּבָרַח עִם מִשְׁפַּחְתּוֹ.

בְּאֶרֶץ מוֹאָב הַמִּשְׁפָּחָה אִבְּדָה אֶת כָּל הַכֶּסֶף שֶׁלָּהּ, וְגַם אֱלִימֶלֶךְ וּשְׁנֵי בָּנָיו מֵתוּ.

"עַמֵּךְ עַמִּי וֵאלֹהַיִךְ אֱלֹהָי"

הִיא הָלְכָה לַשָּׂדֶה כְּדֵי לְלַקֵּט (לֶאֱסוֹף מֵהָרִצְפָּה) שִׁבֳּלִים בּוֹדְדוֹת שֶׁנָּפְלוּ לַקּוֹצְרִים.

רוּת הִיא אִשָּׁה טוֹבַת לֵב, נֶאֱמָנָה וּמְלֵאַת חֶסֶד. הִיא לֹא עָזְבָה אֶת נָעֳמִי בִּשְׁעַת צָרָה וְדָאֲגָה לָהּ לְאֹכֶל בְּאַהֲבָה רַבָּה.
